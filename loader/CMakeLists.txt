cmake_minimum_required(VERSION 3.22)
project(kmod C CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(kmod SHARED src/loader.cpp)
option(KMOD_DEV "Developer loader: allow hot-loading the core from the external dev dir" ON)
if(KMOD_DEV)
  target_compile_definitions(kmod PRIVATE KMOD_DEV=1)
endif()
target_link_libraries(kmod log dl)
target_link_options(kmod PRIVATE -Wl,-z,max-page-size=16384)
