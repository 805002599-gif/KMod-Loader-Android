cmake_minimum_required(VERSION 3.18)
project(param_reader CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# kmod.h: from the repository (core/include) when this example sits in it, otherwise from ./include
if(EXISTS ${CMAKE_SOURCE_DIR}/../../../core/include/kmod.h)
  set(KMOD_INCLUDE ${CMAKE_SOURCE_DIR}/../../../core/include)
else()
  set(KMOD_INCLUDE ${CMAKE_SOURCE_DIR}/include)
endif()
add_library(param_reader SHARED src/main.cpp)
target_include_directories(param_reader PRIVATE ${KMOD_INCLUDE})
target_compile_options(param_reader PRIVATE -fvisibility=hidden -O2)
target_link_options(param_reader PRIVATE -static-libstdc++)
set_target_properties(param_reader PROPERTIES OUTPUT_NAME param_reader PREFIX "lib" LIBRARY_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/native/arm64-v8a)
